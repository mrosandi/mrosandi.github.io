    <!-- SECTION 03 — PROBLEM STATEMENT -->
    <section class="case-section case-problems reveal">
      <div class="case-section-header">
        <span class="case-section-num"><span>03</span> SECTION</span>
        <h3 class="case-section-title">The Problem</h3>
      </div>
      <p class="case-section-desc">
        Community organizations were managing different parts of their community through disconnected platforms. Communication happened in one place, events in another, content somewhere else, and monetization often required additional third-party services.
      </p>
      <div class="case-comparison-infographic">
        <div class="case-infographic-card case-card-before">
          <div class="case-card-header">
            <span class="icon-circle-dot-dashed header-icon"></span>
            <p>This created two connected problems</p>
          </div>
          <ul class="case-tool-list">
            <li>
              <span class="icon-user-star"></span> <span class="tool-name">For Community Managers</span>
                <div class="tool-list"><span class="tool-dot tool-eventbrite"></span>Repeated administrative work across multiple tools</div>
                <div class="tool-list"><span class="tool-dot tool-eventbrite"></span>Member and community data spread across different systems</div>
                <div class="tool-list"><span class="tool-dot tool-eventbrite"></span>Difficulty maintaining consistent communication</div>
                <div class="tool-list"><span class="tool-dot tool-eventbrite"></span>Limited visibility across community activities</div>
            </li>
            <li>
              <span class="icon-user"></span></span><span class="tool-name">For Members</span>
              <div class="tool-list"><span class="tool-dot tool-patreon"></span>Multiple platforms and accounts to manage</div>
              <div class="tool-list"><span class="tool-dot tool-patreon"></span>Notifications and content spread across different channels</div>
              <div class="tool-list"><span class="tool-dot tool-patreon"></span>Friction when moving between community activities</div>
              <div class="tool-list"><span class="tool-dot tool-patreon"></span>Less consistent engagement experience</div>
            </li>
          </ul>
        </div>
        
        <div class="case-infographic-card case-card-problems">
          <div class="case-card-header">
            <span class="icon-circle-alert header-icon"></span>
            <p>Core Problems</p>
          </div>
          <ul class="case-consequence-list">
            <li><span class="list-icon">01</span>
              <div><strong>Fragmented communication</strong>
                <p>Important conversations and updates were distributed across multiple channels.</p>
              </div>
            </li>
            <li><span class="list-icon">02</span>
              <div><strong>Scattered data</strong>
                <p>Member and community information was difficult to manage from one place.</p>
              </div>
            </li>
            <li><span class="list-icon">03</span>
              <div><strong>Poor engagement</strong>
                <p>Fragmented experiences made it harder to create consistent member engagement.</p>
              </div>
            </li>
            <li><span class="list-icon">04</span>
              <div><strong>Limited Ownership</strong>
                <p>Organizations had less control over their community experience when relying on external platforms.</p>
              </div>
            </li>
            <li><span class="list-icon">05</span>
              <div><strong>Limited monetization</strong>
                <p>Events, memberships, and paid content often required separate revenue workflows.</p>
              </div>
            </li>
          </ul>
        </div>
      </div>
    </section>
